# RoomScan — Plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Application iOS qui scanne une pièce au LiDAR, reconnaît murs, ouvertures et meubles, et en affiche une maquette 3D consultable, conservée dans une bibliothèque locale.

**Architecture:** La capture est déléguée à RoomPlan (`RoomCaptureView`) et traduite immédiatement vers un modèle de données maison. Tout le reste — maquette RealityKit, géométrie de caméra, persistance, vignettes — ne connaît que ce modèle, jamais RoomPlan. Le code vit dans un Swift Package local `RoomKit` dont les modules purs se testent sur macOS en ligne de commande ; l'app SwiftUI n'est qu'une couche d'assemblage.

**Tech Stack:** Swift 6.3, SwiftUI, RealityKit, RoomPlan, Core Graphics, Swift Testing, XcodeGen.

## Global Constraints

- Cible de déploiement : **iOS 26.0**. `swift-tools-version: 6.2`, `platforms: [.iOS("26.0"), .macOS("15.0")]`.
- Aucune dépendance externe. XcodeGen (déjà installé en `/opt/homebrew/bin/xcodegen`) sert uniquement à générer le `.xcodeproj`.
- Simulateur de référence : **iPhone 17** (iOS 26.2). Appareil réel de référence : iPhone 15 Pro.
- **Le `.xcodeproj` n'est jamais édité à la main ni commité** — il est généré depuis `project.yml`.
- Textes affichés à l'utilisateur : en français.
- **Piège vérifié :** avec `import SwiftUI` et `import RealityKit` dans le même fichier, le type `Scene` est ambigu. Écrire `var body: some SwiftUI.Scene`.
- **Piège vérifié :** une scène RealityKit non-AR n'a **aucun éclairage par défaut** — sans `DirectionalLight` explicite, l'écran est entièrement noir. Utiliser `SimpleMaterial` (éclairé, donne du relief aux volumes) et non `UnlitMaterial` (silhouettes plates, illisibles en 3D).
- Périmètre : aucune manipulation des meubles (pas de déplacement, rotation, suppression). Voir la spec `docs/superpowers/specs/2026-07-31-scanner-piece-lidar-design.md`.
- Chaque tâche se termine par un commit.

```swift
/// Un mur, décrit par son centre, ses dimensions et son orientation au sol.
public struct Wall: Identifiable, Codable, Hashable, Sendable {
    public let id: UUID
    public let center: SIMD3<Float>
    public let width: Float
    public let height: Float
    /// Rotation autour de l'axe vertical, en radians.

    public init(
        id: UUID = UUID(), center: SIMD3<Float>,
        width: Float, height: Float, yaw: Float
    ) {
        self.id = id
        self.center = center
        self.width = max(width, 0.01)
        self.height = max(height, 0.01)
        self.yaw = yaw
    }

    /// Normale horizontale du mur, dans le plan du sol.
    public var normal: SIMD2<Float> { SIMD2(sin(yaw), cos(yaw)) }

    /// Position du mur projetée au sol.
    public var groundPosition: SIMD2<Float> { SIMD2(center.x, center.z) }
```

```swift
import Foundation
import simd

/// Un meuble reconnu : sa catégorie, son encombrement mesuré et sa pose au sol.
public struct FurniturePiece: Identifiable, Codable, Hashable, Sendable {
    /// Un scan rend parfois des dimensions nulles ou absurdes. Les borner ici
    /// évite qu'un seul objet mal mesuré ne casse le rendu ou le cadrage.
    public static let minimumDimension: Float = 0.05
    public static let maximumDimension: Float = 10

    public let id: UUID
    public let category: FurnitureCategory
    public let center: SIMD3<Float>
    /// Encombrement au sol : largeur puis profondeur.
    public let footprint: SIMD2<Float>
    public let height: Float
    public let yaw: Float
    public let confidence: DetectionConfidence

    public init(
        id: UUID = UUID(), category: FurnitureCategory, center: SIMD3<Float>,
        footprint: SIMD2<Float>, height: Float, yaw: Float,
        confidence: DetectionConfidence
    ) {
        self.id = id
        self.category = category
        self.center = center
        self.footprint = SIMD2(
            FurniturePiece.clamp(footprint.x), FurniturePiece.clamp(footprint.y))
        self.height = FurniturePiece.clamp(height)
        self.yaw = yaw
        self.confidence = confidence
    }

    private static func clamp(_ value: Float) -> Float {
        guard value.isFinite else { return minimumDimension }
        return min(max(abs(value), minimumDimension), maximumDimension)
    }

    public var groundPosition: SIMD2<Float> { SIMD2(center.x, center.z) }

    /// Libellé affiché à la sélection, dimensions arrondies au centimètre.
    public var dimensionsLabel: String {
        let width = Int((footprint.x * 100).rounded())
        let depth = Int((footprint.y * 100).rounded())
        let tall = Int((height * 100).rounded())
        return "\(width) × \(depth) × \(tall) cm"
    }
}
```

Créer `RoomKit/Sources/RoomModel/FloorOutline.swift` :

```swift
import Foundation
import simd

/// Emprise au sol de la pièce. Sert au cadrage de la caméra et à la vignette.
public struct FloorOutline: Codable, Hashable, Sendable {
    public let points: [SIMD2<Float>]

    public init(points: [SIMD2<Float>]) {
        self.points = points
    }

    /// Repli utilisé quand le scanner ne fournit pas de polygone de sol :
    /// le rectangle englobant des murs.
    public init(boundingWalls walls: [Wall]) {
        guard !walls.isEmpty else {
            self.init(points: [])
            return
        }
        var minimum = SIMD2<Float>(repeating: .greatestFiniteMagnitude)
        var maximum = SIMD2<Float>(repeating: -.greatestFiniteMagnitude)
        for wall in walls {
            // Les deux extrémités du mur, obtenues en suivant sa direction.
            let direction = SIMD2(cos(wall.yaw), -sin(wall.yaw)) * (wall.width / 2)
            for end in [wall.groundPosition + direction, wall.groundPosition - direction] {
                minimum = simd_min(minimum, end)
                maximum = simd_max(maximum, end)
            }
        }
        self.init(points: [
            minimum, SIMD2(maximum.x, minimum.y), maximum, SIMD2(minimum.x, maximum.y),
        ])
    }

    public var isEmpty: Bool { points.isEmpty }

    public var center: SIMD2<Float> {
        guard !points.isEmpty else { return .zero }
        return (minimumCorner + maximumCorner) / 2
    }

    public var size: SIMD2<Float> {
        guard !points.isEmpty else { return .zero }
        return maximumCorner - minimumCorner
    }

    /// Rayon du cercle englobant, utilisé pour cadrer la caméra sur la pièce.
    public var boundingRadius: Float {
        guard !points.isEmpty else { return 1 }
        return simd_length(size) / 2
    }

    private var minimumCorner: SIMD2<Float> {
        points.reduce(SIMD2(repeating: .greatestFiniteMagnitude), simd_min)
    }

    private var maximumCorner: SIMD2<Float> {
        points.reduce(SIMD2(repeating: -.greatestFiniteMagnitude), simd_max)
    }
}
```

```swift
public struct ScannedRoom: Codable, Hashable, Sendable {
    public static let currentSchemaVersion = 1

    public let schemaVersion: Int
    public let walls: [Wall]
    public let openings: [Opening]
    public let furniture: [FurniturePiece]
    public let floor: FloorOutline

    public init(
        schemaVersion: Int = ScannedRoom.currentSchemaVersion,
        walls: [Wall], openings: [Opening],
        furniture: [FurniturePiece], floor: FloorOutline
    ) {
        self.schemaVersion = schemaVersion
        self.walls = walls
        self.openings = openings
        self.furniture = furniture
        self.floor = floor.isEmpty ? FloorOutline(boundingWalls: walls) : floor
    }

    /// Un scan sans mur n'est pas exploitable : la pièce n'a pas de forme.
    public var isUsable: Bool { !walls.isEmpty }

    /// Hauteur sous plafond estimée, à partir du mur le plus haut.
    public var ceilingHeight: Float { walls.map(\.height).max() ?? 2.5 }
}
```

```swift
import Foundation

/// Lecture tabulaire du résultat de la reconnaissance : les meubles regroupés
/// par catégorie et comptés.
public enum FurnitureInventory {
    public struct Group: Identifiable, Hashable, Sendable {
        public let category: FurnitureCategory
        public let pieces: [FurniturePiece]

        public var id: String { category.rawValue }
        public var count: Int { pieces.count }

        /// Vrai si au moins un meuble du groupe a été reconnu sans certitude.
        public var hasUncertainDetection: Bool {
            pieces.contains { !$0.confidence.isCertain }
        }
    }

    public static func groups(for room: ScannedRoom) -> [Group] {
        Dictionary(grouping: room.furniture, by: \.category)
            .map { Group(category: $0.key, pieces: $0.value) }
            .sorted { $0.category.displayName < $1.category.displayName }
    }
}
```

```swift
import Foundation
import simd

/// Caméra tournant autour d'un point de visée. Purement géométrique : aucune
/// dépendance au moteur de rendu, ce qui la rend testable sans écran.
public struct OrbitCamera: Equatable, Sendable {
    /// Rester au-dessus du sol évite de voir la maquette par en dessous.
    public static let minElevation: Float = 0.08
    /// Rester en deçà du zénith évite le renversement de l'image au pôle.
    public static let maxElevation: Float = .pi / 2 - 0.02
    public static let minDistance: Float = 0.6
    public static let maxDistance: Float = 40

    public private(set) var target: SIMD3<Float>
    public private(set) var distance: Float
    public private(set) var azimuth: Float
    public private(set) var elevation: Float

    public init(target: SIMD3<Float>, distance: Float, azimuth: Float, elevation: Float) {
        self.target = target
        self.distance = OrbitCamera.clampDistance(distance)
        self.azimuth = OrbitCamera.wrapAngle(azimuth)
        self.elevation = OrbitCamera.clampElevation(elevation)
    }

    /// Position de l'œil, en coordonnées monde.
    public var position: SIMD3<Float> {
        let horizontal = distance * cos(elevation)
        return target + SIMD3(
            horizontal * sin(azimuth),
            distance * sin(elevation),
            horizontal * cos(azimuth))
    }

    public mutating func orbit(deltaAzimuth: Float, deltaElevation: Float) {
        azimuth = OrbitCamera.wrapAngle(azimuth + deltaAzimuth)
        elevation = OrbitCamera.clampElevation(elevation + deltaElevation)
    }

    public mutating func zoom(factor: Float) {
        guard factor > 0, factor.isFinite else { return }
        distance = OrbitCamera.clampDistance(distance * factor)
    }

    /// Déplace le point de visée dans le plan du sol, selon l'orientation
    /// courante de la caméra : « à droite » reste à droite de l'écran.
    public mutating func pan(right: Float, forward: Float) {
        let rightAxis = SIMD3<Float>(cos(azimuth), 0, -sin(azimuth))
        let forwardAxis = SIMD3<Float>(-sin(azimuth), 0, -cos(azimuth))
        target += rightAxis * right + forwardAxis * forward
    }

    public mutating func topDown() {
        elevation = OrbitCamera.maxElevation
    }

    public mutating func recenter(on target: SIMD3<Float>, distance: Float) {
        self.target = target
        self.distance = OrbitCamera.clampDistance(distance)
    }

    private static func clampElevation(_ value: Float) -> Float {
        min(max(value, minElevation), maxElevation)
    }

    private static func clampDistance(_ value: Float) -> Float {
        min(max(value, minDistance), maxDistance)
    }

    private static func wrapAngle(_ value: Float) -> Float {
        var angle = value.truncatingRemainder(dividingBy: 2 * .pi)
        if angle > .pi { angle -= 2 * .pi }
        if angle < -.pi { angle += 2 * .pi }
        return angle
    }
}
```

```swift
import Foundation
import simd
import RoomModel

/// Calcule la distance de recul nécessaire pour que la pièce entière tienne
/// dans le cadre, et la position de départ de la caméra.
public enum FrameFitting {
    /// Marge autour de la pièce, pour ne pas coller aux bords de l'écran.
    public static let margin: Float = 1.15

    public static func distance(
        boundingRadius: Float, verticalFieldOfView: Float, aspectRatio: Float
    ) -> Float {
        let halfVertical = verticalFieldOfView / 2
        // Sur un écran plus étroit que haut, c'est l'horizontale qui contraint.
        let halfHorizontal = atan(tan(halfVertical) * max(aspectRatio, 0.01))
        let limiting = min(halfVertical, halfHorizontal)
        return boundingRadius * margin / max(tan(limiting), 0.01)
    }

    public static func initialCamera(
        for room: ScannedRoom, verticalFieldOfView: Float, aspectRatio: Float
    ) -> OrbitCamera {
        let floorCenter = room.floor.center
        let radius = max(room.floor.boundingRadius, 1)
        return OrbitCamera(
            target: SIMD3(floorCenter.x, room.ceilingHeight / 3, floorCenter.y),
            distance: distance(
                boundingRadius: radius,
                verticalFieldOfView: verticalFieldOfView,
                aspectRatio: aspectRatio),
            azimuth: .pi / 5,
            elevation: .pi / 5)
    }
}
```

```swift
import Foundation
import simd
import RoomModel

/// Détermine quels murs masquer pour qu'on voie l'intérieur de la maquette.
///
/// Un mur situé entre la caméra et le centre de la pièce cache tout ce qu'il y
/// a derrière lui. On le repère sans normale ni orientation : il suffit que la
/// caméra se trouve du côté extérieur du mur.
public enum WallVisibility {
    /// En deçà de cette hauteur d'observation, la question ne se pose plus :
    /// vue du dessus, aucun mur ne gêne.
    public static let overheadThreshold: Float = 0.9

    public static func hiddenWallIDs(
        walls: [Wall], roomCenter: SIMD2<Float>, cameraPosition: SIMD3<Float>
    ) -> Set<UUID> {
        let camera = SIMD2(cameraPosition.x, cameraPosition.z)
        let horizontalDistance = simd_length(camera - roomCenter)

        // Caméra quasiment à la verticale : on regarde par le plafond ouvert.
        guard horizontalDistance > 0.01,
              cameraPosition.y / max(horizontalDistance, 0.01) < 1 / overheadThreshold
        else { return [] }

        return Set(
            walls.filter { wall in
                let outward = wall.groundPosition - roomCenter
                let toCamera = camera - wall.groundPosition
                return simd_dot(outward, toCamera) > 0
            }.map(\.id))
    }
}
```

```swift
import CoreGraphics
import Foundation
import ImageIO
import RoomModel
import UniformTypeIdentifiers
import simd

public enum ThumbnailRendererError: Error, Equatable {
    case contextUnavailable
    case encodingFailed
}

/// Dessine le plan de la pièce vu du dessus : contour des murs et empreintes
/// des meubles. Un dessin 2D plutôt qu'une capture de la scène 3D — c'est
/// déterministe, testable sans écran, et sans dépendance au moteur de rendu.
public enum ThumbnailRenderer {
    public static func pngData(for room: ScannedRoom, pixelSize: Int = 400) throws -> Data {
        let size = max(pixelSize, 32)
        guard let context = CGContext(
            data: nil, width: size, height: size,
            bitsPerComponent: 8, bytesPerRow: 0,
            space: CGColorSpaceCreateDeviceRGB(),
            bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue)
        else { throw ThumbnailRendererError.contextUnavailable }

        context.setFillColor(CGColor(red: 0.97, green: 0.97, blue: 0.98, alpha: 1))
        context.fill(CGRect(x: 0, y: 0, width: size, height: size))

        let transform = makeTransform(for: room, pixelSize: CGFloat(size))
        drawFloor(room.floor, in: context, transform: transform)
        drawWalls(room.walls, in: context, transform: transform)
        drawFurniture(room.furniture, in: context, transform: transform)

        guard let image = context.makeImage() else {
            throw ThumbnailRendererError.contextUnavailable
        }
        let output = NSMutableData()
        guard let destination = CGImageDestinationCreateWithData(
            output, UTType.png.identifier as CFString, 1, nil)
        else { throw ThumbnailRendererError.encodingFailed }

        CGImageDestinationAddImage(destination, image, nil)
        guard CGImageDestinationFinalize(destination) else {
            throw ThumbnailRendererError.encodingFailed
        }
        return output as Data
    }

    /// Convertit les mètres de la pièce en pixels, centrés dans la vignette.
    private static func makeTransform(
        for room: ScannedRoom, pixelSize: CGFloat
    ) -> (SIMD2<Float>) -> CGPoint {
        let extent = max(room.floor.size.x, room.floor.size.y, 1)
        let scale = pixelSize * 0.82 / CGFloat(extent)
        let center = room.floor.center

        return { point in
            let offset = point - center
            return CGPoint(
                x: pixelSize / 2 + CGFloat(offset.x) * scale,
                y: pixelSize / 2 - CGFloat(offset.y) * scale)
        }
    }

    private static func drawFloor(
        _ floor: FloorOutline, in context: CGContext,
        transform: (SIMD2<Float>) -> CGPoint
    ) {
        guard floor.points.count > 2 else { return }
        context.setFillColor(CGColor(red: 1, green: 1, blue: 1, alpha: 1))
        context.beginPath()
        context.move(to: transform(floor.points[0]))
        for point in floor.points.dropFirst() {
            context.addLine(to: transform(point))
        }
        context.closePath()
        context.fillPath()
    }

    private static func drawWalls(
        _ walls: [Wall], in context: CGContext,
        transform: (SIMD2<Float>) -> CGPoint
    ) {
        context.setStrokeColor(CGColor(red: 0.24, green: 0.26, blue: 0.30, alpha: 1))
        context.setLineWidth(3)
        context.setLineCap(.round)

        for wall in walls {
            let direction = SIMD2(cos(wall.yaw), -sin(wall.yaw)) * (wall.width / 2)
            context.move(to: transform(wall.groundPosition - direction))
            context.addLine(to: transform(wall.groundPosition + direction))
        }
        context.strokePath()
    }

    private static func drawFurniture(
        _ furniture: [FurniturePiece], in context: CGContext,
        transform: (SIMD2<Float>) -> CGPoint
    ) {
        context.setFillColor(CGColor(red: 0.35, green: 0.55, blue: 0.78, alpha: 0.75))

        for piece in furniture {
            let halfWidth = piece.footprint.x / 2
            let halfDepth = piece.footprint.y / 2
            let corners: [SIMD2<Float>] = [
                [-halfWidth, -halfDepth], [halfWidth, -halfDepth],
                [halfWidth, halfDepth], [-halfWidth, halfDepth],
            ]
            let rotated = corners.map { corner -> SIMD2<Float> in
                let x = corner.x * cos(piece.yaw) - corner.y * sin(piece.yaw)
                let y = corner.x * sin(piece.yaw) + corner.y * cos(piece.yaw)
                return piece.groundPosition + SIMD2(x, y)
            }

            context.beginPath()
            context.move(to: transform(rotated[0]))
            for corner in rotated.dropFirst() {
                context.addLine(to: transform(corner))
            }
            context.closePath()
            context.fillPath()
        }
    }
}
```

```swift
import Foundation
import RoomModel

/// Une pièce telle qu'elle apparaît dans la bibliothèque, sans charger sa
/// géométrie complète.
public struct RoomEntry: Identifiable, Hashable, Sendable {
    public let metadata: RoomMetadata
    public let thumbnailURL: URL?
    /// Faux si les fichiers sont corrompus ou d'une version inconnue : la pièce
    /// reste visible et supprimable, mais ne s'ouvre pas.
    public let isReadable: Bool

    public var id: UUID { metadata.id }
}

/// Bibliothèque locale : un dossier par pièce, contenant `room.json`,
/// `scan.json` et `thumbnail.png`.
public struct RoomStore: Sendable {
    public let rootDirectory: URL

    public init(rootDirectory: URL) {
        self.rootDirectory = rootDirectory
    }

    public static func documentsStore() throws -> RoomStore {
        let documents = try FileManager.default.url(
            for: .documentDirectory, in: .userDomainMask,
            appropriateFor: nil, create: true)
        let root = documents.appendingPathComponent("Rooms", isDirectory: true)
        try FileManager.default.createDirectory(at: root, withIntermediateDirectories: true)
        return RoomStore(rootDirectory: root)
    }

    // MARK: - Écriture

    @discardableResult
    public func save(_ room: ScannedRoom, named name: String) throws -> RoomMetadata {
        guard room.isUsable else { throw RoomStoreError.unusableScan }

        let metadata = RoomMetadata(name: name)
        let folder = directory(for: metadata.id)
        do {
            try FileManager.default.createDirectory(at: folder, withIntermediateDirectories: true)
            try encoder.encode(room).write(to: folder.appendingPathComponent("scan.json"))
            try encoder.encode(metadata).write(to: folder.appendingPathComponent("room.json"))
            let thumbnail = try ThumbnailRenderer.pngData(for: room)
            try thumbnail.write(to: folder.appendingPathComponent("thumbnail.png"))
        } catch {
            // Ne pas laisser un dossier à moitié écrit derrière soi.
            try? FileManager.default.removeItem(at: folder)
            throw RoomStoreError.writeFailed(error.localizedDescription)
        }
        return metadata
    }

    public func rename(id: UUID, to name: String) throws {
        let url = directory(for: id).appendingPathComponent("room.json")
        guard var metadata = try? decoder.decode(RoomMetadata.self, from: Data(contentsOf: url))
        else { throw RoomStoreError.unreadable(id) }

        metadata.name = name
        metadata.updatedAt = Date()
        do {
            try encoder.encode(metadata).write(to: url)
        } catch {
            throw RoomStoreError.writeFailed(error.localizedDescription)
        }
    }

    public func delete(id: UUID) throws {
        try FileManager.default.removeItem(at: directory(for: id))
    }

    // MARK: - Lecture

    public func entries() -> [RoomEntry] {
        let folders = (try? FileManager.default.contentsOfDirectory(
            at: rootDirectory, includingPropertiesForKeys: nil)) ?? []

        return folders.compactMap(entry(atFolder:))
            .sorted { $0.metadata.createdAt > $1.metadata.createdAt }
    }

    public func load(id: UUID) throws -> ScannedRoom {
        let url = directory(for: id).appendingPathComponent("scan.json")
        guard let data = try? Data(contentsOf: url) else {
            throw RoomStoreError.unreadable(id)
        }
        if let version = schemaVersion(in: data), version > ScannedRoom.currentSchemaVersion {
            throw RoomStoreError.unsupportedSchema(id, version)
        }
        guard let room = try? decoder.decode(ScannedRoom.self, from: data) else {
            throw RoomStoreError.unreadable(id)
        }
        return room
    }

    // MARK: - Détails

    private func directory(for id: UUID) -> URL {
        rootDirectory.appendingPathComponent(id.uuidString, isDirectory: true)
    }

    private func entry(atFolder folder: URL) -> RoomEntry? {
        guard let data = try? Data(contentsOf: folder.appendingPathComponent("room.json")),
              let metadata = try? decoder.decode(RoomMetadata.self, from: data)
        else { return nil }

        let thumbnail = folder.appendingPathComponent("thumbnail.png")
        let readable = (try? load(id: metadata.id)) != nil

        return RoomEntry(
            metadata: metadata,
            thumbnailURL: FileManager.default.fileExists(atPath: thumbnail.path) ? thumbnail : nil,
            isReadable: readable)
    }

    /// Lit la seule version du schéma sans décoder le reste : un fichier écrit
    /// par une version future doit produire un message clair, pas une erreur de
    /// décodage obscure.
    private func schemaVersion(in data: Data) -> Int? {
        let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any]
        return json??["schemaVersion"] as? Int
    }

    private var encoder: JSONEncoder {
        let encoder = JSONEncoder()
        encoder.dateEncodingStrategy = .iso8601
        return encoder
    }

    private var decoder: JSONDecoder {
        let decoder = JSONDecoder()
        decoder.dateDecodingStrategy = .iso8601
        return decoder
    }
}
```

```swift
#if os(iOS)
import Foundation
import RealityKit
import RoomModel
import UIKit
import simd

/// Traduit une pièce scannée en entités RealityKit. Conserve les entités par
/// identifiant pour que la sélection et le masquage des murs puissent les
/// retrouver sans reparcourir la scène.
@MainActor
public final class RoomSceneBuilder {
    public let room: ScannedRoom
    public private(set) var wallEntities: [UUID: Entity] = [:]
    public private(set) var furnitureEntities: [UUID: Entity] = [:]

    private static let wallThickness: Float = 0.06

    public init(room: ScannedRoom) {
        self.room = room
    }

    public func buildScene() -> Entity {
        let root = Entity()
        root.addChild(makeFloor())
        for wall in room.walls {
            let entity = makeWall(wall)
            wallEntities[wall.id] = entity
            root.addChild(entity)
        }
        for opening in room.openings {
            root.addChild(makeOpening(opening))
        }
        for piece in room.furniture {
            let entity = makeFurniture(piece)
            furnitureEntities[piece.id] = entity
            root.addChild(entity)
        }
        root.addChild(makeLighting())
        return root
    }

    // MARK: - Éléments

    private func makeFloor() -> Entity {
        let size = room.floor.size
        let entity = ModelEntity(
            mesh: .generateBox(width: max(size.x, 0.5), height: 0.02, depth: max(size.y, 0.5)),
            materials: [SimpleMaterial(
                color: UIColor(white: 0.88, alpha: 1), isMetallic: false)])
        entity.position = [room.floor.center.x, -0.01, room.floor.center.y]
        entity.name = "floor"
        return entity
    }

    private func makeWall(_ wall: Wall) -> Entity {
        let entity = ModelEntity(
            mesh: .generateBox(
                width: wall.width, height: wall.height, depth: RoomSceneBuilder.wallThickness),
            materials: [SimpleMaterial(
                color: UIColor(white: 0.96, alpha: 1), isMetallic: false)])
        entity.position = wall.center
        entity.orientation = simd_quatf(angle: wall.yaw, axis: [0, 1, 0])
        entity.name = "wall:\(wall.id.uuidString)"
        return entity
    }

    private func makeOpening(_ opening: Opening) -> Entity {
        let color: UIColor = switch opening.kind {
        case .door: UIColor(red: 0.55, green: 0.42, blue: 0.32, alpha: 1)
        case .window: UIColor(red: 0.60, green: 0.80, blue: 0.92, alpha: 1)
        case .opening: UIColor(white: 0.75, alpha: 1)
        }
        // Légèrement plus épais que le mur pour éviter le clignotement de
        // profondeur entre deux surfaces coplanaires.
        let entity = ModelEntity(
            mesh: .generateBox(
                width: opening.width, height: opening.height,
                depth: RoomSceneBuilder.wallThickness * 1.4),
            materials: [SimpleMaterial(color: color, isMetallic: false)])
        entity.position = opening.center
        entity.orientation = simd_quatf(angle: opening.yaw, axis: [0, 1, 0])
        entity.name = "opening:\(opening.id.uuidString)"
        return entity
    }

    private func makeFurniture(_ piece: FurniturePiece) -> Entity {
        let mesh = MeshResource.generateBox(
            width: piece.footprint.x, height: piece.height, depth: piece.footprint.y,
            cornerRadius: 0.01)
        let entity = ModelEntity(
            mesh: mesh,
            materials: [SimpleMaterial(
                color: FurniturePalette.color(for: piece.category), isMetallic: false)])
        // Le centre mesuré est celui du volume ; poser la base au sol.
        entity.position = [piece.center.x, piece.height / 2, piece.center.z]
        entity.orientation = simd_quatf(angle: piece.yaw, axis: [0, 1, 0])
        entity.name = "furniture:\(piece.id.uuidString)"
```

```swift
/// Lecture d'une matrice de transformation : d'où vient l'objet et comment il
/// est tourné au sol. Isolé ici parce que c'est la conversion la plus facile à
/// écrire de travers, et la seule de la traduction qui mérite des tests.
public enum TransformReading {
    public static func position(of transform: simd_float4x4) -> SIMD3<Float> {
        SIMD3(transform.columns.3.x, transform.columns.3.y, transform.columns.3.z)
    }

    /// Angle de rotation autour de l'axe vertical. Lu sur la première colonne,
    /// normalisée : une mise à l'échelle ne doit pas fausser l'angle.
    public static func yaw(of transform: simd_float4x4) -> Float {
        let axis = SIMD2(transform.columns.0.x, transform.columns.0.z)
        guard simd_length(axis) > 0.0001 else { return 0 }
        return atan2(-axis.y, axis.x)
    }
```

```swift
        let furniture = captured.objects.map { object in
            FurniturePiece(
                category: category(of: object.category),
                center: position(of: object.transform),
                // RoomPlan donne largeur, hauteur, profondeur dans cet ordre.
                footprint: SIMD2(object.dimensions.x, object.dimensions.z),
                height: object.dimensions.y,
                yaw: yaw(of: object.transform),
                confidence: confidence(of: object.confidence))
        }

        return ScannedRoom(
            walls: walls, openings: openings, furniture: furniture,
            floor: floorOutline(of: captured, walls: walls))
    }

    // MARK: - Éléments

    private static func translate(
        _ surface: CapturedRoom.Surface, as kind: OpeningKind
    ) -> Opening {
        Opening(
            kind: kind,
            center: position(of: surface.transform),
            width: surface.dimensions.x,
            height: surface.dimensions.y,
            yaw: yaw(of: surface.transform))
    }

    /// RoomPlan fournit une surface de sol depuis iOS 17 ; quand elle manque,
    /// on retombe sur le rectangle englobant des murs.
    private static func floorOutline(
        of captured: CapturedRoom, walls: [Wall]
    ) -> FloorOutline {
        guard let floor = captured.floors.first else {
            return FloorOutline(boundingWalls: walls)
        }
        let center = position(of: floor.transform)
        let angle = yaw(of: floor.transform)
        let halfWidth = floor.dimensions.x / 2
        let halfDepth = floor.dimensions.y / 2

        let corners: [SIMD2<Float>] = [
            [-halfWidth, -halfDepth], [halfWidth, -halfDepth],
            [halfWidth, halfDepth], [-halfWidth, halfDepth],
        ]
        return FloorOutline(points: corners.map { corner in
            SIMD2(
                center.x + corner.x * cos(angle) - corner.y * sin(angle),
                center.z + corner.x * sin(angle) + corner.y * cos(angle))
        })
    }

    private static func position(of transform: simd_float4x4) -> SIMD3<Float> {
        TransformReading.position(of: transform)
    }

    private static func yaw(of transform: simd_float4x4) -> Float {
        TransformReading.yaw(of: transform)
    }

    private static func category(
        of category: CapturedRoom.Object.Category
    ) -> FurnitureCategory {
        switch category {
        case .storage: .storage
        case .refrigerator: .refrigerator
        case .stove: .stove
        case .bed: .bed
        case .sink: .sink
        case .washerDryer: .washerDryer
        case .toilet: .toilet
        case .bathtub: .bathtub
        case .oven: .oven
        case .dishwasher: .dishwasher
        case .table: .table
        case .sofa: .sofa
        case .chair: .chair
        case .fireplace: .fireplace
        case .television: .television
        case .stairs: .stairs
        @unknown default: .unknown(String(describing: category))
        }
    }

    private static func confidence(
        of confidence: CapturedRoom.Confidence
    ) -> DetectionConfidence {
        switch confidence {
        case .high: .high
        case .medium: .medium
        case .low: .low
        @unknown default: .low
        }
    }
}
#endif
```

```swift
import RoomModel
import RoomPlan
import SwiftUI

public enum RoomCaptureAvailability {
    /// Faux sur simulateur et sur tout appareil sans LiDAR.
    public static var isSupported: Bool { RoomCaptureSession.isSupported }
}

public enum RoomCaptureError: Error, LocalizedError {
    case interrupted
    case failed(String)

    public var errorDescription: String? {
        switch self {
        case .interrupted:
            "Le scan a été interrompu."
        case .failed(let reason):
            "Le scan a échoué : \(reason)"
        }
    }
}

/// Écran de capture fourni par RoomPlan : guidage, surlignage temps réel et
/// animation de fin. On n'écrit que la traduction du résultat.
public struct RoomCaptureScreen: UIViewRepresentable {
    private let onFinished: (Result<ScannedRoom, any Error>) -> Void

    public init(onFinished: @escaping (Result<ScannedRoom, any Error>) -> Void) {
        self.onFinished = onFinished
    }

    public func makeUIView(context: Context) -> RoomCaptureView {
        let view = RoomCaptureView(frame: .zero)
        view.delegate = context.coordinator
        context.coordinator.start(view)
        return view
    }

    public func updateUIView(_ view: RoomCaptureView, context: Context) {}

    public func makeCoordinator() -> Coordinator {
        Coordinator(onFinished: onFinished)
    }

    public static func dismantleUIView(_ view: RoomCaptureView, coordinator: Coordinator) {
        coordinator.stop(view)
    }

    @MainActor
    public final class Coordinator: NSObject, RoomCaptureViewDelegate {
        private let onFinished: (Result<ScannedRoom, any Error>) -> Void
        private var isRunning = false

        init(onFinished: @escaping (Result<ScannedRoom, any Error>) -> Void) {
            self.onFinished = onFinished
        }

        func start(_ view: RoomCaptureView) {
            guard !isRunning else { return }
            isRunning = true
            view.captureSession.run(configuration: RoomCaptureSession.Configuration())
        }

        func stop(_ view: RoomCaptureView) {
            guard isRunning else { return }
            isRunning = false
            view.captureSession.stop()
        }

        /// Demander le post-traitement : c'est lui qui produit la pièce propre.
        public func captureView(
            shouldPresent roomDataForProcessing: CapturedRoomData, error: (any Error)?
        ) -> Bool {
            if let error {
                onFinished(.failure(RoomCaptureError.failed(error.localizedDescription)))
                return false
            }
            return true
        }

        public func captureView(didPresent processedResult: CapturedRoom, error: (any Error)?) {
            isRunning = false
            if let error {
                onFinished(.failure(RoomCaptureError.failed(error.localizedDescription)))
            } else {
                onFinished(.success(CapturedRoomTranslator.translate(processedResult)))
            }
        }
    }
}
```

```swift
struct LibraryView: View {

    @State private var isScanning = false

                    row(for: entry)
                }

                    .foregroundStyle(.secondary)

                        .font(.caption)
                        .foregroundStyle(.orange)
                }
            }
        }
        .swipeActions(edge: .leading) {
            Button("Renommer") {
                newName = entry.metadata.name
                renaming = entry
            }
            .tint(.blue)
        }
    }

        if let url = entry.thumbnailURL, let image = UIImage(contentsOfFile: url.path) {
            Image(uiImage: image)
                .resizable()
                .scaledToFit()
                .frame(width: 54, height: 54)
                .clipShape(RoundedRectangle(cornerRadius: 8))
        } else {
            RoundedRectangle(cornerRadius: 8)
                .fill(.quaternary)
                .frame(width: 54, height: 54)
        }

            self.store = store

        } catch {
            failure = error.localizedDescription
        }
    }

    private func delete(at offsets: IndexSet) {
        guard let store else { return }
        for index in offsets {
            try? store.delete(id: entries[index].id)
        }
        reload()
    }

    private func commitRename() {
        guard let store, let entry = renaming else { return }
        do {
            try store.rename(id: entry.id, to: newName)
        } catch {
            failure = error.localizedDescription
        }
        renaming = nil
        reload()
    }
```
